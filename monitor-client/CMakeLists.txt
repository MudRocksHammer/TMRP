cmake_minimum_required(VERSION 3.21)
project(TmrpMonitorClient VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network WebSockets)
qt_add_executable(tmrp-monitor-client src/main.cpp src/diagram.cpp src/diagram.h src/telemetrystream.cpp src/telemetrystream.h)
target_link_libraries(tmrp-monitor-client PRIVATE Qt6::Widgets Qt6::Network Qt6::WebSockets)
if(MSVC)
  target_compile_options(tmrp-monitor-client PRIVATE /W4 /utf-8)
else()
  target_compile_options(tmrp-monitor-client PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS tmrp-monitor-client RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(test_monitor_diagram tests/test_diagram.cpp src/diagram.cpp src/diagram.h)
  target_include_directories(test_monitor_diagram PRIVATE src)
  target_link_libraries(test_monitor_diagram PRIVATE Qt6::Widgets Qt6::Test)
  if(MSVC)
    target_compile_options(test_monitor_diagram PRIVATE /utf-8)
  endif()
  add_test(NAME monitor_diagram COMMAND test_monitor_diagram)
  set_tests_properties(monitor_diagram PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  qt_add_executable(test_monitor_stream tests/test_telemetrystream.cpp src/telemetrystream.cpp src/telemetrystream.h)
  target_include_directories(test_monitor_stream PRIVATE src)
  target_link_libraries(test_monitor_stream PRIVATE Qt6::WebSockets Qt6::Test)
  if(MSVC)
    target_compile_options(test_monitor_stream PRIVATE /utf-8)
  endif()
  add_test(NAME monitor_stream COMMAND test_monitor_stream)
  set_tests_properties(monitor_stream PROPERTIES TIMEOUT 45)
endif()
