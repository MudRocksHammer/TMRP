cmake_minimum_required(VERSION 3.21)
project(TmrpDeviceSimulator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Sql Network)
find_package(unofficial-mosquitto CONFIG QUIET)
if(TARGET unofficial::mosquitto::mosquitto)
  set(MQTT_TARGET unofficial::mosquitto::mosquitto)
else()
  find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h REQUIRED)
  find_library(MOSQUITTO_LIBRARY NAMES mosquitto REQUIRED)
  add_library(mosquitto_dependency UNKNOWN IMPORTED)
  set_target_properties(mosquitto_dependency PROPERTIES
    IMPORTED_LOCATION "${MOSQUITTO_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${MOSQUITTO_INCLUDE_DIR}")
  set(MQTT_TARGET mosquitto_dependency)
endif()
add_library(simulator_core src/deviceengine.cpp src/mqtttransport.cpp src/deviceprofile.cpp)
target_include_directories(simulator_core PUBLIC src)
target_link_libraries(simulator_core PUBLIC Qt6::Core Qt6::Sql Qt6::Network PRIVATE ${MQTT_TARGET})
add_library(simulator_ui src/mainwindow.cpp src/fleetwindow.cpp)
target_link_libraries(simulator_ui PUBLIC simulator_core Qt6::Widgets)
add_executable(tmrp-device-simulator src/main.cpp)
target_link_libraries(tmrp-device-simulator PRIVATE simulator_ui)
foreach(target simulator_core simulator_ui tmrp-device-simulator)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test_deviceengine tests/test_deviceengine.cpp)
  target_link_libraries(test_deviceengine PRIVATE simulator_core Qt6::Test)
  add_test(NAME device_engine COMMAND test_deviceengine)
  add_executable(test_window tests/test_window.cpp)
  target_link_libraries(test_window PRIVATE simulator_ui Qt6::Test)
  add_test(NAME control_window COMMAND test_window)
  add_executable(test_fleet tests/test_fleet.cpp)
  target_link_libraries(test_fleet PRIVATE simulator_ui Qt6::Test)
  add_test(NAME fleet_management COMMAND test_fleet)
  set_tests_properties(fleet_management PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  set_tests_properties(control_window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(device_engine control_window PROPERTIES TIMEOUT 30)
endif()
if(MSVC)
  # The QtTest sources also contain Japanese literals and must use UTF-8.
  add_compile_options(/utf-8)
  if(BUILD_TESTING)
    foreach(target test_deviceengine test_window test_fleet)
      target_compile_options(${target} PRIVATE /utf-8)
    endforeach()
  endif()
endif()
install(TARGETS tmrp-device-simulator RUNTIME DESTINATION bin)
